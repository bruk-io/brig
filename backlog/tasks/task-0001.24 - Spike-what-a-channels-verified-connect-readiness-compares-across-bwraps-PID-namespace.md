## Description

<!-- SECTION:DESCRIPTION:BEGIN -->
decision-0002 item 2 says a channel is ready when a connect succeeds *and* is confirmed to be the jailed process's own listener, never when the path exists (old SPEC decision-185: a co-tenant symlinked a reserved socket path to a listener outside the jail, and a path-existence check reported it ready). It does not say how that confirmation works here. Under --unshare-all the supervisor is PID 1 in its own PID namespace, so a peer pid read off the socket on the host (SO_PEERCRED on Linux, LOCAL_PEERPID on macOS) is a host pid, not directly comparable to anything the jail reports about itself. This spike answers what the readiness check compares against what, measured on real bwrap, before the readiness task is planned.

## Spike result (2026-10-19)

**Answer.** Readiness compares the PID namespace of the process that called listen() on the socket that answered the connect (SO_PEERPIDFD on the connected socket, then readlink /proc/<pid>/ns/pid, re-checked against the pidfd) against the PID namespace of the jail's in-jail PID 1 (bwrap's child whose NSpid ends in 1). It never compares the path, and it never compares uid. Seatbelt (stated, not measured): the peer's LOCAL_PEERTOKEN audit token (pid plus pidversion) and an ancestry walk to the Seatbelt supervisor, which is weaker.

**Recorded in decision-0005** (backlog/decisions/decision-0005 - Channel-readiness-compares-the-listen-callers-PID-namespace-with-the-jails-own-inside-a-same-uid-trust-boundary.md), accepted by @bruk 2026-10-19. The spike itself judged no new decision was needed; the skeptics' findings and the owner's threat-model ruling made one necessary. It refines, not reverses, decision-0002 item 2.

**Why the alternatives lose.**
- lstat/O_NOFOLLOW plus S_ISSOCK rejects symlinks (2a, 2b) but accepted (2c), a co-tenant's real socket bound at the reserved path; it also races between the check and connect(), which has no O_NOFOLLOW.
- Peer pid == supervisor pid is too narrow: the listener is normally a descendant (peer pid 11, supervisor 10).
- Walking the PPid chain brings back reparenting and pid-reuse races.
- NSpid alone shows the peer is in some nested namespace, not this one.
- /proc/net/unix inode matching needs a /proc/*/fd scan or sock_diag and still yields no namespace identity.
- uid cannot tell them apart: the outside listener had the jail's uid in every run (0 as root, 1000 as uid 1000).
- The NS_GET_PARENT walk (accepts nested namespaces, case 4) adds an ioctl loop and no security; not the default.

"""Throwaway probe for TASK-0001.24: what a channel's verified-connect readiness compares.

Runs as the "host" inside a privileged Linux container with real bwrap. Builds a jail the
way compile_supervisor_launch orders it (binds, state/channel binds, virtual fs, then
--unshare-all --die-with-parent --new-session --as-pid-1 --), with the per-jail channel
directory bind-mounted writable at /chan, and a PID-1 "supervisor" that forks the real
listener as its own child.

Readiness under test: connect() to the reserved path succeeds AND the connected peer's pid
namespace (SO_PEERPIDFD -> pid -> /proc/<pid>/ns/pid, re-checked against the pidfd) equals
the jail's pid namespace (read from the in-jail PID 1, found as bwrap's child on the host).
"""

import errno
import json
import os
import select
import signal
import socket
import struct
import subprocess
import sys
import time

SO_PEERPIDFD = getattr(socket, "SO_PEERPIDFD", 77)
BASE = os.environ.get("SPIKE_BASE", "/run/spike24")
CHAN = f"{BASE}/chan"          # per-jail channel directory, host side
OUTSIDE = f"{BASE}/outside"    # a host directory the jail cannot write
RESERVED = f"{CHAN}/svc.sock"  # reserved path, host side; /chan/svc.sock inside the jail

# The jail's payload: PID 1 forks the real listener (a descendant, like a supervised command),
# then reaps. MODE selects what the jailed program does with its reserved path.
JAIL_PROGRAM = r"""
import os, socket, sys, time, subprocess
mode = sys.argv[1]
def serve(path):
    s = socket.socket(socket.AF_UNIX, socket.SOCK_STREAM); s.bind(path); s.listen(8)
    while True:
        c, _ = s.accept(); c.close()
pid = os.fork()
if pid == 0:
    if mode == "own":
        serve(os.environ.get("SPIKE_BASE", "/run/spike24") + "/chan/svc.sock")
    elif mode == "plant-symlink":
        # The jail itself plants the symlink: the target string is resolved by whoever
        # connects -- on the host, that is the host's own /run/spike24/outside/host.sock.
        os.symlink(os.environ.get("SPIKE_BASE", "/run/spike24") + "/outside/host.sock", os.environ.get("SPIKE_BASE", "/run/spike24") + "/chan/svc.sock")
        time.sleep(3600)
    elif mode == "nested-pidns":
        # A listener inside a pid namespace nested under the jail's own.
        os.execvp("unshare", ["unshare", "--user", "--map-root-user", "--pid", "--fork", sys.executable, "-c",
            "import socket,sys\ns=socket.socket(socket.AF_UNIX);s.bind(sys.argv[1]);s.listen(8)\n"
            "while True:\n c,_=s.accept();c.close()", os.environ.get("SPIKE_BASE", "/run/spike24") + "/chan/svc.sock"])
    os._exit(0)
while True:
    try: os.wait()
    except ChildProcessError: time.sleep(3600)
"""


def log(label: str, **fields: object) -> None:
    """Print one result line."""
    print(f"{label:<44} " + " ".join(f"{k}={v}" for k, v in fields.items()), flush=True)


def reset_dirs() -> None:
    """Recreate the channel and outside directories empty."""
    subprocess.run(["rm", "-rf", BASE], check=True)
    os.makedirs(CHAN)
    os.makedirs(OUTSIDE)


def outside_listener(path: str) -> subprocess.Popen[bytes]:
    """Start a host process (outside any jail) listening at `path`."""
    code = (
        "import socket,sys\ns=socket.socket(socket.AF_UNIX);s.bind(sys.argv[1]);s.listen(8)\n"
        "print('up',flush=True)\nwhile True:\n c,_=s.accept();c.close()"
    )
    proc = subprocess.Popen([sys.executable, "-c", code, path], stdout=subprocess.PIPE)
    assert proc.stdout is not None and proc.stdout.readline() == b"up\n"
    return proc


def start_jail(mode: str) -> subprocess.Popen[bytes]:
    """Start bwrap with the jail program, flags ordered like compile_supervisor_launch."""
    argv = [
        "bwrap",
        "--ro-bind", "/", "/",
        "--bind", CHAN, CHAN,
        "--proc", "/proc", "--dev", "/dev", "--tmpfs", "/sys",
        "--unshare-all", "--die-with-parent", "--new-session", "--as-pid-1",
        "--",
        sys.executable, "-c", JAIL_PROGRAM, mode,
    ]
    return subprocess.Popen(argv)


def children(pid: int) -> list[int]:
    """Return `pid`'s direct children, by scanning /proc for PPid."""
    out = []
    for entry in os.listdir("/proc"):
        if entry.isdigit():
            try:
                with open(f"/proc/{entry}/status") as f:
                    for line in f:
                        if line.startswith("PPid:") and int(line.split()[1]) == pid:
                            out.append(int(entry))
            except OSError:
                pass
    return out


def nspid(pid: int) -> str:
    """Return the NSpid line of /proc/<pid>/status."""
    with open(f"/proc/{pid}/status") as f:
        return next(l.split(":", 1)[1].strip() for l in f if l.startswith("NSpid:"))


def jail_pid_ns(bwrap_pid: int) -> tuple[int, str]:
    """Find the in-jail PID 1 (bwrap's child in a new pid ns) and return (host pid, ns link)."""
    deadline = time.monotonic() + 5
    while time.monotonic() < deadline:
        for child in children(bwrap_pid):
            if nspid(child).split()[-1] == "1":
                return child, os.readlink(f"/proc/{child}/ns/pid")
        time.sleep(0.05)
    raise RuntimeError("no in-jail pid 1 found")


def wait_for(predicate, timeout: float = 5.0) -> None:
    """Poll until `predicate()` is true."""
    deadline = time.monotonic() + timeout
    while not predicate():
        if time.monotonic() > deadline:
            raise TimeoutError
        time.sleep(0.05)


def naive_ready(path: str) -> str:
    """The rejected design: ready when connect succeeds."""
    s = socket.socket(socket.AF_UNIX, socket.SOCK_STREAM)
    try:
        s.connect(path)
        return "READY"
    except OSError as e:
        return f"not-ready({errno.errorcode[e.errno]})"
    finally:
        s.close()


def lstat_ready(path: str) -> str:
    """The path-based alternative: refuse a symlink, require S_ISSOCK, then connect."""
    import stat
    try:
        st = os.lstat(path)
    except FileNotFoundError:
        return "not-ready(ENOENT)"
    if stat.S_ISLNK(st.st_mode):
        return "rejected(symlink)"
    if not stat.S_ISSOCK(st.st_mode):
        return "rejected(not-a-socket)"
    return naive_ready(path)


NS_GET_PARENT = 0xB702


def ns_is_or_under(peer_pid: int, jail_ns: str) -> tuple[bool, str]:
    """Walk the peer's pid namespace up with NS_GET_PARENT until it equals the jail's or ends."""
    import fcntl
    fd = os.open(f"/proc/{peer_pid}/ns/pid", os.O_RDONLY)
    chain = []
    try:
        while True:
            here = os.readlink(f"/proc/self/fd/{fd}")
            chain.append(here)
            if here == jail_ns:
                return True, ">".join(chain)
            try:
                parent = fcntl.ioctl(fd, NS_GET_PARENT)
            except OSError as e:
                chain.append(errno.errorcode[e.errno])
                return False, ">".join(chain)
            os.close(fd)
            fd = parent
    finally:
        os.close(fd)


def verified_ready(path: str, jail_ns: str, allow_nested: bool = False) -> str:
    """The proposed design: connect, then compare the peer's pid namespace with the jail's."""
    s = socket.socket(socket.AF_UNIX, socket.SOCK_STREAM)
    try:
        try:
            s.connect(path)
        except OSError as e:
            return f"not-ready({errno.errorcode[e.errno]})"
        pid, uid, gid = struct.unpack("3i", s.getsockopt(socket.SOL_SOCKET, socket.SO_PEERCRED, 12))
        pidfd = s.getsockopt(socket.SOL_SOCKET, SO_PEERPIDFD)
        try:
            peer_ns = os.readlink(f"/proc/{pid}/ns/pid")
            # Pin the pid read above to the very process the kernel recorded: if it exited
            # and the pid was reused, the pidfd's fdinfo shows Pid: -1 (or a mismatch).
            with open(f"/proc/self/fdinfo/{pidfd}") as f:
                fd_pid = next(int(l.split()[1]) for l in f if l.startswith("Pid:"))
        except OSError as e:
            return f"rejected(peer-gone {errno.errorcode[e.errno]})"
        finally:
            os.close(pidfd)
        detail = f"peer_host_pid={pid} peer_uid={uid} pidfd_pid={fd_pid} peer_ns={peer_ns}"
        if fd_pid != pid:
            return f"rejected(pid-reused) {detail}"
        if allow_nested:
            ok, chain = ns_is_or_under(pid, jail_ns)
            detail += f" ns_chain={chain}"
            if not ok:
                return f"rejected(peer-outside-jail) {detail}"
        elif peer_ns != jail_ns:
            return f"rejected(peer-outside-jail) {detail}"
        return f"READY {detail}"
    finally:
        s.close()


def run_case(name: str, jail_mode: str, setup=None, expect_listener: bool = True) -> None:
    """Run one scenario and print all three checks."""
    reset_dirs()
    host = outside_listener(f"{OUTSIDE}/host.sock")
    extra = setup() if setup else None
    jail = start_jail(jail_mode)
    try:
        sup_host_pid, jail_ns = jail_pid_ns(jail.pid)
        if jail_mode == "own" or jail_mode == "nested-pidns" or jail_mode == "plant-symlink":
            wait_for(lambda: os.path.lexists(RESERVED))
            time.sleep(0.3)
        print(f"--- {name}")
        log("  host: bwrap pid / its ns/pid", bwrap=jail.pid, ns=os.readlink(f"/proc/{jail.pid}/ns/pid"))
        log("  host: in-jail PID 1 (bwrap child)", host_pid=sup_host_pid, NSpid=repr(nspid(sup_host_pid)), ns=jail_ns)
        log("  reserved path lstat", kind=describe(RESERVED))
        log("  naive (connect only)", result=naive_ready(RESERVED))
        log("  lstat/S_ISSOCK + connect", result=lstat_ready(RESERVED))
        log("  verified (connect + peer pid-ns == jail)", result=verified_ready(RESERVED, jail_ns))
        log("  verified, descendant ns allowed", result=verified_ready(RESERVED, jail_ns, allow_nested=True))
    finally:
        jail.send_signal(signal.SIGKILL)
        jail.wait()
        host.kill()
        host.wait()
        if extra is not None:
            extra.kill()
            extra.wait()


def describe(path: str) -> str:
    """Describe what sits at `path` without following it."""
    import stat
    try:
        st = os.lstat(path)
    except FileNotFoundError:
        return "absent"
    if stat.S_ISLNK(st.st_mode):
        return f"symlink->{os.readlink(path)}"
    if stat.S_ISSOCK(st.st_mode):
        return "socket"
    if stat.S_ISREG(st.st_mode):
        return "regular-file"
    return oct(st.st_mode)


def main() -> None:
    """Run every scenario."""
    print("kernel", os.uname().release, "python", sys.version.split()[0])
    print("host (probe) pid", os.getpid(), "ns", os.readlink("/proc/self/ns/pid"))

    # (1) the jail's own listener (a child of the in-jail PID 1) at the reserved path.
    run_case("(1) jail's own listener at reserved path", "own")

    # (2a) the jail itself plants a symlink at the reserved path to a host listener.
    run_case("(2a) jail plants symlink -> host listener", "plant-symlink")

    # (2b) a co-tenant on the host plants the symlink (old SPEC decision-185's shape).
    def cotenant_symlink():
        os.symlink(f"{OUTSIDE}/host.sock", RESERVED)
    run_case("(2b) host co-tenant plants symlink -> host listener", "idle", setup=cotenant_symlink)

    # (2c) a host co-tenant binds a real socket at the reserved path itself (no symlink).
    run_case("(2c) host co-tenant binds real socket at reserved path", "idle",
             setup=lambda: outside_listener(RESERVED))

    # (3a) a stale socket inode: bound and closed, nothing listening.
    def stale_socket():
        s = socket.socket(socket.AF_UNIX); s.bind(RESERVED); s.close()
    run_case("(3a) stale socket file, nothing listening", "idle", setup=stale_socket)

    # (3b) a regular file at the reserved path.
    run_case("(3b) regular file at reserved path", "idle",
             setup=lambda: open(RESERVED, "w").close())

    # (3c) a dangling symlink at the reserved path.
    run_case("(3c) dangling symlink", "idle",
             setup=lambda: os.symlink(f"{OUTSIDE}/nothing.sock", RESERVED))

    # (4) a listener in a pid namespace nested under the jail's (fail-closed edge).
    run_case("(4) jail listener inside a nested pid ns", "nested-pidns")


if __name__ == "__main__":
    main()
